
    # The shop's one-file product page -> backend SSR (per-product title,
    # description, canonical, og:*, Product JSON-LD). EXACT match: every other
    # file under /shop/ (shop.js, shop.css, the checkout and thank-you dirs)
    # keeps being served statically. Without this, all products share one set
    # of meta and read as duplicates of each other.
    location = /shop/product.html {
        proxy_pass http://rc-backend:8000;
        proxy_read_timeout 60s;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
