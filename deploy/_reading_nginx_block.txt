
    # Homepage + the six daily chart pages -> backend SSR. Each renders through
    # js/chart-shell.js, so the backend cannot ship a copy of that module (see
    # page_ssr._ssr_reading_summary); it ships the plain data summary the shell
    # overwrites on load.
    #
    # EXACT matches (`location = `), every one. `location /` keeps serving every
    # asset beneath these paths -- chart.js, chart.css, the images -- and the
    # bare-root match cannot swallow anything else.
    location = / {
        proxy_pass http://rc-backend:8000;
        proxy_read_timeout 60s;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
